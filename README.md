const area = document.getElementById('hearts');

function criarCoracao() {
    const coracao = document.createElement('span');
    coracao.className = 'heart';
    coracao.textContent = ['♡', '♥', '💕', '💗'][
        Math.floor(Math.random() * 4)
    ];

    coracao.style.left = Math.random() * 100 + '%';
    coracao.style.fontSize = (14 + Math.random() * 20) + 'px';
    coracao.style.animationDuration = (6 + Math.random() * 7) + 's';

    area.appendChild(coracao);

    setTimeout(() => coracao.remove(), 13000);
}

function criarCoracoes() {
    for (let i = 0; i < 22; i++) {
        setTimeout(criarCoracao, i * 100);
    }
}

setInterval(criarCoracao, 700);
